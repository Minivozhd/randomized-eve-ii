cmake_minimum_required(VERSION 3.27)
project(pe2randomizer VERSION 1.3.1 DESCRIPTION "Parasite Eve 2 Randomizer" LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake --preset release")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
option(PE2_BUILD_GUI "Build the wxWidgets application" ON)
option(PE2_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(PE2_ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers (Clang/GCC)" OFF)
option(PE2_BUNDLE_DEPENDENCIES "Copy dependent libraries into the installed macOS app" ON)
set(PE2_RELEASE_DATE "24.10.2026")
set(PE2_COPYRIGHT_YEARS "2023-2026")
configure_file(cmake/BuildInfo.hpp.in generated/BuildInfo.hpp @ONLY)

find_package(Threads REQUIRED)
include(cmake/Sources.cmake)
function(pe2_target_defaults target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/src")
    if(PE2_ENABLE_SANITIZERS)
        if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "PE2_ENABLE_SANITIZERS requires Clang or GCC")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(PE2_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra)
        if(PE2_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()

# Compile the same implementation once for the app and command-line tests.
add_library(pe2_core STATIC ${PE2_CORE_SOURCES})
pe2_target_defaults(pe2_core)
target_compile_definitions(pe2_core PUBLIC PE2_ENABLE_EXPERIMENTAL_BOSSES=0)
target_compile_definitions(pe2_core PUBLIC PE2_ENABLE_DEBUG_WEAPONS=0)
target_link_libraries(pe2_core PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(pe2_core PUBLIC ws2_32)
endif()

if(PE2_BUILD_GUI)
    find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
    include(cmake/FindQrencode.cmake)
    if(WIN32)
        enable_language(RC)
        configure_file(cmake/icon.rc.in generated/icon.rc @ONLY)
        add_executable(pe2randomizer WIN32 src/main.cpp ${PE2_UI_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/generated/icon.rc")
    elseif(APPLE)
        add_executable(pe2randomizer MACOSX_BUNDLE src/main.cpp ${PE2_UI_SOURCES})
        set_target_properties(pe2randomizer PROPERTIES
            MACOSX_BUNDLE_BUNDLE_NAME "PE2 Randomizer"
            MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
            MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
            MACOSX_BUNDLE_GUI_IDENTIFIER "com.pe2randomizer.app")
    else()
        add_executable(pe2randomizer src/main.cpp ${PE2_UI_SOURCES})
    endif()
    pe2_target_defaults(pe2randomizer)
    target_include_directories(pe2randomizer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(pe2randomizer PRIVATE pe2_core wxWidgets::wxWidgets PE2::Qrencode)
    include(cmake/RuntimeResources.cmake)
    include(cmake/Install.cmake)
endif()

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
