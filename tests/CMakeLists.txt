# Release builds keep assertions enabled in every test target.
function(pe2_add_test name source label)
    add_executable(pe2_${name} ${source})
    pe2_target_defaults(pe2_${name})
    target_link_libraries(pe2_${name} PRIVATE pe2_core)
    if(MSVC)
        target_compile_options(pe2_${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(pe2_${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND pe2_${name})
    set_tests_properties(${name} PROPERTIES LABELS "rom-free;${label}" TIMEOUT 120)
endfunction()
